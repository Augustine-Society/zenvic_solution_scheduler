-- 1. Workers Directory Table
CREATE TABLE IF NOT EXISTS workers (
    duty_number INTEGER PRIMARY KEY,
    full_name TEXT NOT NULL,
    department TEXT DEFAULT 'General',
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Forward Weekly Schedules Table
CREATE TABLE IF NOT EXISTS schedules (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    duty_number INTEGER NOT NULL REFERENCES workers(duty_number) ON DELETE CASCADE,
    shift_date DATE NOT NULL,
    shift_type TEXT NOT NULL CHECK (shift_type IN ('On-Site', 'WFH', 'Rest Day')),
    start_time TIME,
    end_time TIME,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    CONSTRAINT unique_worker_shift_date UNIQUE (duty_number, shift_date)
);

-- 3. Punch Logs Table (Actual Attendance Records)
CREATE TABLE IF NOT EXISTS punch_logs (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    duty_number INTEGER NOT NULL REFERENCES workers(duty_number) ON DELETE CASCADE,
    log_date DATE NOT NULL,
    status TEXT NOT NULL CHECK (status IN ('On Duty', 'WFH', 'Clocked Out')),
    clock_in TIME NOT NULL,
    clock_out TIME,
    total_hours NUMERIC(5, 2) DEFAULT 0.00,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    CONSTRAINT unique_worker_daily_punch UNIQUE (duty_number, log_date)
);
